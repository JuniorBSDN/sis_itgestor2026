let itensRequeridos = []; // Gerencia a lista visual de itens (Teclado, Mouse, etc) function adicionarItem() { const input = document.getElementById("inputItem"); const valor = input.value.trim(); if (valor && !itensRequeridos.includes(valor)) { itensRequeridos.push(valor); renderizarItens(); input.value = ""; } } function removerItem(item) { itensRequeridos = itensRequeridos.filter(i => i !== item); renderizarItens(); } function renderizarItens() { const container = document.getElementById("listaItens"); container.innerHTML = itensRequeridos.map(item => ` ${item} `).join(''); } // FUNÇÃO DE BUSCA async function buscarAtivo() { const idBusca = document.getElementById("inputBusca").value; // Corrigido para o ID do campo de busca if (!idBusca) return alert("Digite um ID para buscar"); try { const res = await fetch('/api/ativos'); const data = await res.json(); // A lógica de busca depende da estrutura do seu retorno de API const ativo = data.ativos.find(a => a.id_ativo == idBusca); if (ativo) { preencherFormulario(ativo); document.getElementById("btnExcluir").classList.remove("hidden"); } else { alert("Ativo não encontrado."); } } catch (err) { console.error("Erro na busca:", err); } } function preencherFormulario(ativo) { document.getElementById("id_ativo").value = ativo.id_ativo || ""; document.getElementById("usuario").value = ativo.usuario || ""; document.getElementById("setor").value = ativo.setor || ""; document.getElementById("equipamento").value = ativo.equipamento || ""; document.getElementById("modelo").value = ativo.modelo || ""; document.getElementById("ns_serial").value = ativo.ns_serial || ""; document.getElementById("patrimonio").value = ativo.patrimonio || ""; document.getElementById("status_fisico").value = ativo.status_fisico || ""; document.getElementById("hardware").value = ativo.hardware || ""; document.getElementById("software").value = ativo.software || ""; document.getElementById("id_suporte").value = ativo.id_suporte || ""; document.getElementById("ipv4").value = ativo.ipv4 || ""; document.getElementById("nome_computador").value = ativo.nome_computador || ""; document.getElementById("hostname").value = ativo.hostname || ""; document.getElementById("obs").value = ativo.obs || ""; itensRequeridos = ativo.itens_requeridos || []; renderizarItens(); } // INTERCEPTAR O ENVIO DO FORMULÁRIO document.getElementById("formAtivo").addEventListener("submit", async (e) => { e.preventDefault(); // Impede o recarregamento da página const dados = { id_ativo: document.getElementById("id_ativo").value, usuario: document.getElementById("usuario").value, setor: document.getElementById("setor").value, equipamento: document.getElementById("equipamento").value, modelo: document.getElementById("modelo").value, ns_serial: document.getElementById("ns_serial").value, patrimonio: document.getElementById("patrimonio").value, hardware: document.getElementById("hardware").value, software: document.getElementById("software").value, id_suporte: document.getElementById("id_suporte").value, ipv4: document.getElementById("ipv4").value, nome_computador: document.getElementById("nome_computador").value, hostname: document.getElementById("hostname").value, status_fisico: document.getElementById("status_fisico").value, obs: document.getElementById("obs").value, itens_requeridos: itensRequeridos, data_atualizacao: new Date().toLocaleString('pt-BR') }; try { const res = await fetch('/api/ativos', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(dados) }); if (res.ok) { alert("✅ Ativo salvo com sucesso!"); limparFormulario(); } } catch (err) { alert("❌ Erro ao salvar: " + err.message); } }); function limparFormulario() { document.getElementById("formAtivo").reset(); itensRequeridos = []; renderizarItens(); document.getElementById("btnExcluir").classList.add("hidden"); } // FUNÇÃO PARA GERAR TXT (Simulação local) function gerarTXTLocal() { const nomeArquivo = `ativo_${document.getElementById("id_ativo").value || 'sem_id'}.txt`; const conteudo = `ID: ${document.getElementById("id_ativo").value}\nUsuário: ${document.getElementById("usuario").value}\nSetor: ${document.getElementById("setor").value}`; const blob = new Blob([conteudo], { type: 'text/plain' }); const link = document.createElement('a'); link.href = window.URL.createObjectURL(blob); link.download = nomeArquivo; link.click(); }
